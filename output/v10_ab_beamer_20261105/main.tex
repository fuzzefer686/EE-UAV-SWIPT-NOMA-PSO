\begin{frame}{Công suất động cơ: ba thành phần và các mốc tính lại}
\footnotesize
\setlength{\abovedisplayskip}{4pt}
\setlength{\belowdisplayskip}{4pt}
\[
T_r(v)=\frac{\sqrt{(mg)^2+(\rho S_{FP}v^2/2)^2}}{n_r},\qquad
P_f(v)=P_{\rm profile}+P_{\rm induced}+P_{\rm parasite}.
\]
\begin{align*}
P_{\rm profile}&=n_r\frac{\delta}{8}\left(\frac{T_r}{c_T\rho A}+3v^2\right)\sqrt{\frac{T_r\rho c_s^2A}{c_T}},\\[-2pt]
P_{\rm induced}&=n_r(1+c_f)T_r\sqrt{\sqrt{\frac{T_r^2}{4\rho^2A^2}+\frac{v^4}{4}}-\frac{v^2}{2}},\qquad
P_{\rm parasite}=\frac{n_r}{2}d_0\rho c_sAv^3.
\end{align*}

\begin{frame}{EE và hai mức QoS dùng chung cho mọi thuật toán}
\small
\[
\mathrm{EE}_{\rm UAV}=\frac{\sum_{u=1}^{5}\sum_{n=1}^{20}D_{u,n}}
{P_f(v)\sum_{n=1}^{20}d_n/v+P_f(0)\sum_{n=1}^{20}h_n+\sum_{n=1}^{20}P_{{\rm RF},n}T_{{\rm RF},n}}.
\]
\begin{columns}[T,onlytextwidth]
\column{.49\textwidth}
\begin{block}{Yêu cầu về dữ liệu}
Mỗi user phải thỏa cả hai mức:
\[\sum_n D_{u,n}\geq L,\qquad D_{u,n}\geq q\quad\forall u,n.\]
\footnotesize $L$: bit/user/nhiệm vụ. $q$: bit/user/slot phục vụ, không phải bit/s. Ngưỡng tổng bit/slot đặt bằng 0.
\end{block}
\column{.49\textwidth}
\begin{block}{Một chu kỳ phục vụ và di chuyển}
\footnotesize Tại waypoint $n$: EH/UL trong $h/2$, rồi RF/DL trong $h/2$; tiếp theo bay đoạn $d_n$ trong $d_n/v$.
\[E_{\rm total}\leq17\,000\;\mathrm{J}.\]
20 lần phục vụ và 20 đoạn bay; điểm đến cuối chỉ kết thúc hành trình.
\end{block}
\end{columns}
\notevi{EH cấp RF lý tưởng. EE dùng tổng năng lượng tiêu thụ cruise + hover + RF; năng lượng pin lý tưởng chỉ gồm cruise + hover.}
\src{\texttt{evaluateTrajectoryV10}, \texttt{addV10EnergyMetrics}; cấu hình MAT.}
\end{frame}

\begin{frame}{Các giới hạn cần giữ khi sử dụng kết quả}
\small
\begin{description}
\item[Tìm kiếm] Ba scenario seed, một algorithm seed; B/IPSO còn tăng EE trong nhiều lượt ở cuối 1.000 vòng. Chưa đo độ ổn định qua nhiều khởi tạo thuật toán.
\item[Ngân sách] Mutation và polishing làm số lần đánh giá khác nhau. Kết quả chưa phải so sánh trên cùng thời gian hoặc cùng số evaluation.
\item[Năng lượng] RF chiếm tối đa khoảng $2{,}43\times10^{-8}$ tổng tải trong campaign. Propulsion chi phối mẫu số; BS/EH vẫn ảnh hưởng dữ liệu đạt được.
\item[Bay] B giữ Y đơn điệu và có đoạn tiến gần 1 m. Chưa mô hình hóa gia tốc, năng lượng rẽ, mạch/PA hay giới hạn chiều dài $D_{\max}$ như bài Drones.
\item[Dịch vụ] Tối đa toàn bộ bit/J có thể phục vụ một user nhiều hơn ngưỡng $L$. Khi quyết định hover hay BS, nên xem thêm bit/user và fairness.
\end{description}
\src{Mã evaluator/optimizer; dữ liệu cuối và báo cáo audit campaign 29/10/2026.}
\end{frame}
